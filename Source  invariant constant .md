import numpy as np

# Stability classification
def classify_stability(alpha, eta, w_a):
    lam = alpha + eta + w_a
    if abs(lam) < 1: return "Stable"
    elif abs(lam) > 1: return "Unstable"
    else: return "Neutral"

# Parameter sweep
alphas = np.linspace(-2,2,30); etas = np.linspace(-2,2,30); w_as = np.linspace(-1,1,30)
stable, unstable, neutral = [], [], []
for alpha in alphas:
    for eta in etas:
        for w_a in w_as:
            status = classify_stability(alpha, eta, w_a)
            if status == "Stable": stable.append([alpha, eta, w_a])
            elif status == "Unstable": unstable.append([alpha, eta, w_a])
            else: neutral.append([alpha, eta, w_a])
stable = np.array(stable); unstable = np.array(unstable); neutral = np.array(neutral)
